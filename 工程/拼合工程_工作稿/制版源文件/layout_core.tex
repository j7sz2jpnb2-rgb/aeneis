    BoldFont=EBGaramond12-Bold.ttf
  ]{EBGaramond12-Regular.ttf}

    Path=/usr/share/fonts/truetype/ebgaramond/,

    BoldFont=EBGaramond12-Bold.ttf
  ]{EBGaramond12-Regular.ttf}
}{%

    \setmainfont{EB Garamond}

  }%
}

\newcommand{\verseline}[2]{%
  \par\noindent
  \begingroup
  \fontsize{10.2pt}{13.65pt}\selectfont
  \hangindent=3.55em\hangafter=1
  % 为提高扫读速度，所有页的原诗行号统一置于正文左侧。
  \llap{\makebox[2.75em][r]{\sanscjk\fontsize{7.4pt}{9pt}\selectfont\color{softgray}#1}\hspace{0.8em}}%
  #2\par
  \endgroup
  \penalty150
}

  \markboth{#1}{#1}
  \thispagestyle{chapterstart}
  \vspace*{19mm}
  \begin{center}
    {\latinfont\fontsize{10}{12}\selectfont\color{softgray}\addfontfeatures{LetterSpace=12} #2\par}
    \vspace{6mm}
    {\titlecjk\bfseries\fontsize{22}{28}\selectfont\color{accent}#1\par}
    \vspace{5mm}
  \end{center}
  \vspace{11mm}
}

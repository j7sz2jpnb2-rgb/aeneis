\newcommand{\platefile}[1]{\csname platefile#1\endcsname}

\makeatletter
\newcommand{\manualtocchapter}[2]{%
  \addtocontents{toc}{\protect\contentsline{chapter}{#1}{\thepage}{#2}\protected@file@percent}%
}
\makeatother

\newcommand{\mappage}[3]{%
  \clearpage
  \phantomsection
  \pdfbookmark[0]{#2}{map.#3}%
  \manualtocchapter{#2}{map.#3.0}%
  \markboth{#2}{#2}%
  \thispagestyle{chapterstart}%
  \AddToShipoutPictureBG*{\AtPageLowerLeft{%
    % 地图 PDF 默认按右页定位；左页整体左移 4 mm，使装订侧仍留 18 mm。
    \ifodd\value{page}\else\hspace*{-4mm}\fi
    \includegraphics[width=\paperwidth]{#1}%
  }}%
  \null
  \clearpage
}

\newcommand{\platecontent}[3]{%
  \thispagestyle{chapterstart}
  \pdfbookmark[1]{卷首画：#3}{plate.book.#1}%
  \markboth{#3}{#3}%
  \vspace*{6mm}
  \begin{center}
    {\latinfont\fontsize{9.5}{11}\selectfont\color{softgray}\addfontfeatures{LetterSpace=11} #2\par}
    \vspace{5.5mm}
    \includegraphics[width=121mm,height=166mm,keepaspectratio]{\platefile{#1}}\par
    \vspace{5mm}
    {\titlecjk\bfseries\fontsize{11.5}{15}\selectfont\color{accent}\platetitle{#1}\par}
    \vspace{2.2mm}
    {\sanscjk\fontsize{7.1}{9.5}\selectfont\color{softgray}
      \platecredit\par}
  \end{center}
}

\renewcommand{\bookchapter}[3]{%
  \ifstrequal{#3}{1}{%
    \mappage{strabo-world.pdf}{地图 1：斯特拉波所述的世界}{strabo}%
    \mappage{aeneas-voyage.pdf}{地图 2：从特洛伊到拉提乌姆}{voyage}%
  }{}%
  \ifstrequal{#3}{7}{%
    \mappage{latium-tiber.pdf}{地图 3：拉提乌姆与台伯河}{latium}%
  }{}%
  \clearpage
  \phantomsection
  \pdfbookmark[0]{#1}{book.#3}%
  \platecontent{#3}{#2}{\platetitle{#3}}%
  \clearpage
  \phantomsection
  \pdfbookmark[1]{正文}{body.book.#3}%
  \manualtocchapter{#1}{body.book.#3.1}%
  \markboth{#1}{#1}%
  \thispagestyle{chapterstart}%
  \vspace*{19mm}
  \begin{center}
    {\latinfont\fontsize{10}{12}\selectfont\color{softgray}\addfontfeatures{LetterSpace=12} #2\par}
    \vspace{6mm}
    {\titlecjk\bfseries\fontsize{22}{28}\selectfont\color{accent}#1\par}
    \vspace{5mm}
  \end{center}
  \vspace{11mm}
}
